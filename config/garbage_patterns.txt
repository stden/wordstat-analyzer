# Regex-паттерны для фильтрации мусора
# Один паттерн на строку
# Строки с # игнорируются

# Слишком короткие
^.{1,3}$

# Только цифры
^\d+$

# Телефоны
\+?\d[\d\s\-]{9,}

# Даты в разных форматах
\d{2}[\.\/]\d{2}[\.\/]\d{2,4}

# Email-подобное
\S+@\S+\.\S+

# URL
https?://
www\.

# Повторяющиеся символы (ааааа, !!!!)
(.)\1{4,}

# Только спецсимволы
^[\W\d]+$
